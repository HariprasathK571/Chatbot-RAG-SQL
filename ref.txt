from sqlalchemy import create_engine, text
from langchain_community.utilities import SQLDatabase

# --- Create engine ---
engine = create_engine(
    "mssql+pyodbc://user:password@host:1433/mydb?driver=ODBC+Driver+17+for+SQL+Server",
    pool_size=2,
    max_overflow=0,
    pool_recycle=1800,
)

db = SQLDatabase(engine=engine)

# --- Run first query (opens a connection) ---
with engine.connect() as conn:
    result = conn.execute(text("SELECT 1 AS test"))
    print("First query result:", result.fetchall())

# --- Check pool state ---
print("Pool checked out connections before dispose:", engine.pool.checkedout())

# --- Dispose connections ---
engine.dispose()
print("✅ Engine disposed. All pooled connections closed.")

# --- Run another query (this will re-create connections) ---
with engine.connect() as conn:
    result = conn.execute(text("SELECT 2 AS test"))
    print("Second query result:", result.fetchall())

print("Pool checked out connections after re-connect:", engine.pool.checkedout())

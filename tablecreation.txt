📘 Customers
CREATE TABLE customers (
    customer_id        BIGSERIAL PRIMARY KEY,
    full_name          VARCHAR(100) NOT NULL,
    email              VARCHAR(150) UNIQUE,
    created_at         TIMESTAMP DEFAULT CURRENT_TIMESTAMP);
📘 Accounts
CREATE TABLE accounts (
    account_id     BIGSERIAL PRIMARY KEY,
    customer_id    BIGINT REFERENCES customers(customer_id),
    account_type   VARCHAR(20) CHECK (account_type IN ('Savings', 'Current')),
    balance        NUMERIC(15,2) NOT NULL,
    status         VARCHAR(20) CHECK (status IN ('Active', 'Dormant', 'Closed')),
    opened_date    DATE NOT NULL);
📘 Transactions
CREATE TABLE transactions (
    transaction_id     BIGSERIAL PRIMARY KEY,
    account_id         BIGINT REFERENCES accounts(account_id),
    transaction_date   TIMESTAMP NOT NULL,
    amount             NUMERIC(15,2) NOT NULL,
    transaction_type   VARCHAR(10) CHECK (transaction_type IN ('Debit', 'Credit')),
    merchant_name      VARCHAR(100),
    transaction_status VARCHAR(20) CHECK (transaction_status IN ('Success', 'Failed', 'Pending'))
);
📘 Transfers (for internal fund movement)
CREATE TABLE transfers (
    transfer_id        BIGSERIAL PRIMARY KEY,
    from_account_id    BIGINT REFERENCES accounts(account_id),
    to_account_id      BIGINT REFERENCES accounts(account_id),
    transfer_amount    NUMERIC(15,2),
    transfer_date      TIMESTAMP,
    transfer_status    VARCHAR(20)
);
